# dss-core: Kern (Hintergrundprogramm), ohne Fenster
# P13: obsevents (Adapter OBS → Ereignisfluss)
# P14 b3: actionservice (Aktionssystem im Kern); angelegt nur mit DSS_CORE_ACTIONS
qt_add_executable(dss-core main.cpp coreservice.h coreservice.cpp obsevents.h obsevents.cpp
  actionservice.h actionservice.cpp)
target_link_libraries(dss-core PRIVATE dss_common dss_obs dss_events dss_actions dss_warnings Qt6::Network)
target_compile_definitions(dss-core PRIVATE DSS_CORE_ACTIONS=$<BOOL:${DSS_CORE_ACTIONS}>)
# Watchdog von systemd (P10a, Punkt 41, Schalter DSS_CORE_WATCHDOG)
target_compile_definitions(dss-core PRIVATE DSS_CORE_WATCHDOG=$<BOOL:${DSS_CORE_WATCHDOG}>
  DSS_CORE_IMPORT_THREAD=$<BOOL:${DSS_CORE_IMPORT_THREAD}>)
# Grösse neuer Bild- und Videoquellen aus der Datei (P09, Schalter DSS_MEDIA_SIZE)
if(DSS_MEDIA_SIZE)
  target_link_libraries(dss-core PRIVATE dss_media)
  target_compile_definitions(dss-core PRIVATE DSS_MEDIA_SIZE=1)
endif()
# Version im Einsatz (P10a): <Präfix>/bin
if(DSS_INSTALLED_EDITION)
  install(TARGETS dss-core RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
# Ton der Videos über PipeWire (P12b b3, Schalter DSS_CORE_AUDIO)
if(DSS_CORE_AUDIO)
  target_link_libraries(dss-core PRIVATE dss_pwoutput)
  target_compile_definitions(dss-core PRIVATE DSS_CORE_AUDIO=1)
else()
  target_compile_definitions(dss-core PRIVATE DSS_CORE_AUDIO=0)
endif()
# Overlay-Server (P10, Schalter DSS_CORE_OVERLAY_SERVER). Die Seite (web/) liegt in Arbeit im
# Repo, im Einsatz im Installationsordner (wie Symbole und Schriften, P10a).
if(DSS_CORE_OVERLAY_SERVER)
  target_link_libraries(dss-core PRIVATE dss_server)
  if(DSS_INSTALLED_EDITION)
    set(DSS_WEB_DIR "${DSS_INSTALL_FULL_SHARE_DIR}/web")
    install(FILES ${PROJECT_SOURCE_DIR}/web/overlay.html ${PROJECT_SOURCE_DIR}/web/overlay.js
            DESTINATION ${DSS_INSTALL_SHARE_DIR}/web)
  else()
    set(DSS_WEB_DIR "${PROJECT_SOURCE_DIR}/web")
  endif()
  target_compile_definitions(dss-core PRIVATE DSS_CORE_OVERLAY_SERVER=1 DSS_WEB_DIR="${DSS_WEB_DIR}")
else()
  target_compile_definitions(dss-core PRIVATE DSS_CORE_OVERLAY_SERVER=0)
endif()
# Anbindung an Twitch (P13, Schalter DSS_CORE_TWITCH)
if(DSS_CORE_TWITCH)
  target_link_libraries(dss-core PRIVATE dss_twitch)
  target_compile_definitions(dss-core PRIVATE DSS_CORE_TWITCH=1)
else()
  target_compile_definitions(dss-core PRIVATE DSS_CORE_TWITCH=0)
endif()
# P12b b4: Meldungen der Seiten (…/clock) an den Mischer; braucht Ton und Overlay-Server
if(DSS_CORE_AUDIO AND DSS_CORE_OVERLAY_SERVER)
  target_sources(dss-core PRIVATE videoaudio.h videoaudio.cpp)
endif()
